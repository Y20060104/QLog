cmake_minimum_required(VERSION 3.22)
project(QLogDiagnostics LANGUAGES CXX)
find_package(Threads REQUIRED)

if(NOT TARGET QLog::qlog)
    if(NOT DEFINED BUILD_TESTING)
        set(BUILD_TESTING OFF CACHE BOOL "Build QLog functional tests")
    endif()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../.." qlog-source)
endif()

add_executable(QLogComponentBenchmark component_benchmark.cpp)
target_link_libraries(QLogComponentBenchmark PRIVATE QLog::qlog)
qlog_enable_warnings(QLogComponentBenchmark)
add_executable(QLogLatencyProbe latency_probe.cpp)
target_link_libraries(QLogLatencyProbe PRIVATE QLog::qlog)
qlog_enable_warnings(QLogLatencyProbe)
if(WIN32)
    target_compile_definitions(QLogComponentBenchmark PRIVATE NOMINMAX)
    target_compile_definitions(QLogLatencyProbe PRIVATE NOMINMAX)
endif()

set(QLOG_BQLOG_SOURCE_DIR "" CACHE PATH "Fixed BQLog source for optional reference probes")
set(QLOG_BQLOG_LIBRARY "" CACHE FILEPATH "Matching Release BQLog static library")
if(QLOG_BQLOG_SOURCE_DIR OR QLOG_BQLOG_LIBRARY)
    if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
        message(FATAL_ERROR "The optional prebuilt BQLog reference probes require a Release build")
    endif()
    if(NOT EXISTS "${QLOG_BQLOG_SOURCE_DIR}/src/bq_log/log/layout.h" OR
       NOT EXISTS "${QLOG_BQLOG_LIBRARY}")
        message(FATAL_ERROR "Provide both the fixed BQLog source tree and matching static library")
    endif()
    foreach(kind IN ITEMS ComponentBenchmark LatencyProbe)
        if(kind STREQUAL "ComponentBenchmark")
            set(probe_source component_benchmark.cpp)
        else()
            set(probe_source latency_probe.cpp)
        endif()
        add_executable(BQLog${kind} ${probe_source})
        target_compile_features(BQLog${kind} PRIVATE cxx_std_20)
        target_compile_definitions(BQLog${kind} PRIVATE QLOG_DIAG_BQLOG)
        target_include_directories(BQLog${kind} PRIVATE
            "${QLOG_BQLOG_SOURCE_DIR}/include" "${QLOG_BQLOG_SOURCE_DIR}/src")
        target_link_libraries(BQLog${kind} PRIVATE "${QLOG_BQLOG_LIBRARY}" Threads::Threads)
        if(WIN32)
            target_link_libraries(BQLog${kind} PRIVATE dbghelp)
            target_compile_definitions(BQLog${kind} PRIVATE NOMINMAX)
        endif()
    endforeach()
endif()
